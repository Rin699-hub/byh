using System;
using System.Drawing;
using System.IO;
using System.Windows.Forms;
using Microsoft.Web.WebView2.Core;
using Microsoft.Web.WebView2.WinForms;

namespace MyBrowserApp
{
    public partial class Form1 : Form
    {
        // 画面上部レイアウト
        private TableLayoutPanel topPanel = new TableLayoutPanel();

        // 画面パーツ
        private Button btnBack = new Button();
        private Button btnForward = new Button();
        private Button btnReload = new Button();
        private TextBox txtUrl = new TextBox();
        private Button btnGo = new Button();
        private Button btnSettings = new Button();

        // Webブラウザ本体
        private WebView2 webView = new WebView2();

        // アプリの設定値（デフォルト）
        private string homeUrl = "https://www.google.com";
        private string searchEngineUrl = "https://www.google.com/search?q=";
        private string settingsFilePath = "settings.txt";

        public Form1()
        {
            LoadSettings(); // 起動時に設定ファイルを読み込み

            this.Size = new Size(1100, 750);
            this.Text = "マイ Web ブラウザ";

            SetupUI();
            InitBrowser();
        }

        // --- ⚙️ 設定ファイルの読み込み ---
        private void LoadSettings()
        {
            try
            {
                if (File.Exists(settingsFilePath))
                {
                    string[] lines = File.ReadAllLines(settingsFilePath);
                    if (lines.Length >= 1 && !string.IsNullOrWhiteSpace(lines[0]))
                        homeUrl = lines[0].Trim();
                    if (lines.Length >= 2 && !string.IsNullOrWhiteSpace(lines[1]))
                        searchEngineUrl = lines[1].Trim();
                }
            }
            catch
            {
                // 読み込み失敗時はデフォルト設定を使用
            }
        }

        // --- ⚙️ 設定ファイルの保存 ---
        private void SaveSettings()
        {
            try
            {
                string[] lines = { homeUrl.Trim(), searchEngineUrl.Trim() };
                File.WriteAllLines(settingsFilePath, lines);
            }
            catch (Exception ex)
            {
                MessageBox.Show("設定の保存に失敗しました: " + ex.Message, "エラー", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void SetupUI()
        {
            topPanel.Dock = DockStyle.Top;
            topPanel.Height = 45;
            topPanel.BackColor = Color.FromArgb(230, 230, 230);
            topPanel.Padding = new Padding(5);

            topPanel.ColumnCount = 6;
            topPanel.RowCount = 1;

            topPanel.ColumnStyles.Add(new ColumnStyle(SizeType.Absolute, 45F));
            topPanel.ColumnStyles.Add(new ColumnStyle(SizeType.Absolute, 45F));
            topPanel.ColumnStyles.Add(new ColumnStyle(SizeType.Absolute, 55F));
            topPanel.ColumnStyles.Add(new ColumnStyle(SizeType.Percent, 100F));
            topPanel.ColumnStyles.Add(new ColumnStyle(SizeType.Absolute, 60F));
            topPanel.ColumnStyles.Add(new ColumnStyle(SizeType.Absolute, 60F));

            void StyleButton(Button btn, string text)
            {
                btn.Text = text;
                btn.Dock = DockStyle.Fill;
                btn.BackColor = Color.White;
                btn.ForeColor = Color.Black;
                btn.Font = new Font("メイリオ", 9F, FontStyle.Bold);
                btn.UseVisualStyleBackColor = false;
                btn.Margin = new Padding(2);
            }

            StyleButton(btnBack, "←");
            btnBack.Click += (s, e) => { if (webView.CanGoBack) webView.GoBack(); };

            StyleButton(btnForward, "→");
            btnForward.Click += (s, e) => { if (webView.CanGoForward) webView.GoForward(); };

            StyleButton(btnReload, "更新");
            btnReload.Click += (s, e) => webView.Reload();

            txtUrl.Font = new Font("メイリオ", 10F);
            txtUrl.Dock = DockStyle.Fill;
            txtUrl.Margin = new Padding(3, 4, 3, 3);
            txtUrl.KeyDown += (s, e) =>
            {
                if (e.KeyCode == Keys.Enter)
                {
                    NavigateToUrl();
                    e.SuppressKeyPress = true;
                }
            };

            StyleButton(btnGo, "移動");
            btnGo.Click += (s, e) => NavigateToUrl();

            StyleButton(btnSettings, "設定");
            btnSettings.Click += BtnSettings_Click;

            topPanel.Controls.Add(btnBack, 0, 0);
            topPanel.Controls.Add(btnForward, 1, 0);
            topPanel.Controls.Add(btnReload, 2, 0);
            topPanel.Controls.Add(txtUrl, 3, 0);
            topPanel.Controls.Add(btnGo, 4, 0);
            topPanel.Controls.Add(btnSettings, 5, 0);

            webView.Dock = DockStyle.Fill;

            this.Controls.Add(webView);
            this.Controls.Add(topPanel);
        }

        private async void InitBrowser()
        {
            await webView.EnsureCoreWebView2Async(null);

            webView.SourceChanged += (s, e) =>
            {
                if (webView.Source != null)
                {
                    txtUrl.Text = webView.Source.ToString();
                }
            };

            webView.CoreWebView2.DocumentTitleChanged += (s, e) =>
            {
                this.Text = $"{webView.CoreWebView2.DocumentTitle} - マイ Web ブラウザ";
            };

            webView.Source = new Uri(homeUrl);
        }

        private void NavigateToUrl()
        {
            string input = txtUrl.Text.Trim();
            if (string.IsNullOrEmpty(input)) return;

            // http:// または https:// から始まる場合は直接開く
            if (input.StartsWith("http://", StringComparison.OrdinalIgnoreCase) ||
                input.StartsWith("https://", StringComparison.OrdinalIgnoreCase))
            {
                webView.Source = new Uri(input);
            }
            // ドットが含まれ、かつスペースを含まない（例: yahoo.co.jp）場合は https:// を補完して開く
            else if (input.Contains(".") && !input.Contains(" "))
            {
                webView.Source = new Uri("https://" + input);
            }
            // それ以外（単語入力やスペース区切り検索など）は設定された検索エンジンで検索
            else
            {
                string searchUrl = $"{searchEngineUrl}{Uri.EscapeDataString(input)}";
                webView.Source = new Uri(searchUrl);
            }
        }

        // --- ⚙️ 設定画面ダイアログ ---
        private void BtnSettings_Click(object? sender, EventArgs e)
        {
            Form configForm = new Form
            {
                Text = "ブラウザ設定",
                Size = new Size(400, 320),
                StartPosition = FormStartPosition.CenterParent,
                FormBorderStyle = FormBorderStyle.FixedDialog,
                MaximizeBox = false,
                MinimizeBox = false
            };

            Label lblHome = new Label { Text = "🏠 起動時に開くページ (URL):", Left = 20, Top = 20, AutoSize = true };
            TextBox txtHome = new TextBox { Text = homeUrl, Left = 20, Top = 45, Width = 340 };

            Label lblSearch = new Label { Text = "🔍 検索エンジン:", Left = 20, Top = 85, AutoSize = true };
            ComboBox cmbSearch = new ComboBox { Left = 20, Top = 110, Width = 340, DropDownStyle = ComboBoxStyle.DropDownList };
            cmbSearch.Items.AddRange(new object[] { "Google", "Yahoo! JAPAN", "Bing", "DuckDuckGo" });

            // 現在設定されている検索エンジンを確実に判別して初期選択
            string currentSearchLower = searchEngineUrl.ToLower();
            if (currentSearchLower.Contains("yahoo")) cmbSearch.SelectedIndex = 1;
            else if (currentSearchLower.Contains("bing")) cmbSearch.SelectedIndex = 2;
            else if (currentSearchLower.Contains("duckduckgo")) cmbSearch.SelectedIndex = 3;
            else cmbSearch.SelectedIndex = 0; // デフォルト: Google

            Button btnClearCache = new Button { Text = "🗑️ 閲覧データ (キャッシュ/Cookie) を削除", Left = 20, Top = 155, Width = 340, Height = 35 };
            btnClearCache.Click += async (s, ev) =>
            {
                if (webView.CoreWebView2 != null)
                {
                    await webView.CoreWebView2.Profile.ClearBrowsingDataAsync(
                        CoreWebView2BrowsingDataKinds.CacheStorage |
                        CoreWebView2BrowsingDataKinds.Cookies |
                        CoreWebView2BrowsingDataKinds.BrowsingHistory
                    );
                    MessageBox.Show("閲覧データを削除しました！", "完了", MessageBoxButtons.OK, MessageBoxIcon.Information);
                }
            };

            Button btnSave = new Button { Text = "保存して閉じる", Left = 230, Top = 220, Width = 130, Height = 35, DialogResult = DialogResult.OK };
            btnSave.Click += (s, ev) =>
            {
                homeUrl = txtHome.Text.Trim();

                // 選択された検索エンジンのURLをセット
                switch (cmbSearch.SelectedIndex)
                {
                    case 0: searchEngineUrl = "https://www.google.com/search?q="; break;
                    case 1: searchEngineUrl = "https://search.yahoo.co.jp/search?p="; break;
                    case 2: searchEngineUrl = "https://www.bing.com/search?q="; break;
                    case 3: searchEngineUrl = "https://duckduckgo.com/?q="; break;
                }

                // 設定を保存
                SaveSettings();

                // アプリを終了
                configForm.Close();
                this.Close();
            };

            configForm.Controls.Add(lblHome);
            configForm.Controls.Add(txtHome);
            configForm.Controls.Add(lblSearch);
            configForm.Controls.Add(cmbSearch);
            configForm.Controls.Add(btnClearCache);
            configForm.Controls.Add(btnSave);

            configForm.ShowDialog(this);
        }

        private void Form1_Load(object sender, EventArgs e)
        {

        }
    }
}
